let speedInterval = null;
api.mapkey('<Ctrl-Alt-s>', 'Toggle 3x video speed', function() {
    const vid = document.querySelector('video');

    if (!vid) {
        api.Front.showStatus('No video found');
        return;
    }

    if (speedInterval === null) {
        speedInterval = setInterval(() => {
            vid.playbackRate = 3.0;
        }, 900);

        vid.playbackRate = 3.0;
        api.Front.showStatus('3x speed: ON');
    } else {
        clearInterval(speedInterval);
        speedInterval = null;

        vid.playbackRate = 2.0;
        api.Front.showStatus('3x speed: OFF');
    }
});
